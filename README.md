# MyBatis 기반 비회원 게시판
- 여러 종류의 문서를 `doc_management` 테이블에서 통합 관리
- 문서에 첨부된 여러 파일을 `doc_management` 테이블에서 관리

## 게시판 유형에 따른 문서 번호 형식
- 이미지 파일 : BRD-IMG-YYYYMMdd
- PDF 파일 : BRD-PDF-YYYYMMdd
- 통합 파일 : BRD-FILE-YYYYMMdd

※ 실무에서는 메뉴별로 게시판 테이블을 분리하고 문서 번호를 부여하여 관리했지만, 이번 프로젝트에서는 게시판 유형별로 테이블을 분리하지 않고 문서 번호를 통해 게시판 유형을 구분합니다.

# PostgreSQL DB 테이블 생성 SQL
```
CREATE TABLE doc_management(
	id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
	doc_seq varchar(30) NOT NULL UNIQUE,
	is_deleted boolean NOT NULL DEFAULT FALSE,
	created_at timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP,
	updated_at timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE attachment_file(
	id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
	doc_id bigint NOT NULL,
	file_name varchar(200) NOT NULL,
	file_url varchar(200) NOT NULL,
	file_type varchar(50) NOT NULL,
	is_deleted boolean NOT NULL DEFAULT FALSE,
	created_at timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP,
	updated_at timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP,
	
	CONSTRAINT FK_ATTACHMENT_FILE_DOC 
	FOREIGN KEY (doc_id) REFERENCES doc_management(id)
);

CREATE TABLE board(
	id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
	doc_id bigint NOT NULL,
	title varchar(50) NOT NULL,
	content text NOT NULL,
	author varchar(30) NOT NULL,
	password varchar(200) NOT NULL,
	created_at timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP,
	updated_at timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP,
	
	CONSTRAINT FK_BOARD_DOC
	FOREIGN KEY (doc_id) REFERENCES doc_management(id)
);

CREATE TABLE tag(
	id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
	tag varchar(50) NOT NULL UNIQUE,
	use_cnt int NOT NULL DEFAULT 0,
	created_at timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP,
	updated_at timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP,
	
	CONSTRAINT CK_TAG_USE_CNT CHECK (use_cnt >= 0)
);

CREATE TABLE board_tag(
	id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
	doc_id bigint NOT NULL,
	tag_id bigint NOT NULL,
	created_at timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP,
	
	CONSTRAINT FK_BOARD_TAG_DOC
	FOREIGN KEY (doc_id) REFERENCES doc_management(id),
	
	CONSTRAINT FK_BOARD_TAG_TAG
	FOREIGN KEY (tag_id) REFERENCES tag(id),
	
	CONSTRAINT UK_BOARD_TAG UNIQUE (doc_id, tag_id)
);
```
